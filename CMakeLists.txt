function(engine_options target kind)
  target_compile_definitions(${target} PRIVATE ${BASE_DEFINITIONS})
  target_include_directories(${target} PRIVATE ${BASE_INCLUDES})
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_definitions(${target} PRIVATE CINTERFACE= WIN32 _WINDOWS _CRT_SECURE_NO_DEPRECATE _WINSOCK_DEPRECATED_NO_WARNINGS "$<$<CONFIG:Debug>:_DEBUG>" "$<$<NOT:$<CONFIG:Debug>>:NDEBUG>")
    target_compile_options(${target} PRIVATE "$<$<AND:$<COMPILE_LANGUAGE:C,CXX>,$<CONFIG:Debug>>:/Od;/RTC1;/W3;/fp:except>" "$<$<AND:$<COMPILE_LANGUAGE:C,CXX>,$<NOT:$<CONFIG:Debug>>>:/O2;/W4>" "$<$<COMPILE_LANGUAGE:CXX>:/std:c++20;/GR-;/EHs-c-;/Zc:strictStrings>")
    if(kind STREQUAL "native")
      target_compile_options(${target} PRIVATE /fp:strict /Oi-)
    elseif(kind STREQUAL "renderer")
      target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:/fp:fast;/Oi>")
    endif()
  else()
    separate_arguments(extra NATIVE_COMMAND "${AFTERSHOCK_EXTRA_FLAGS}")
    target_compile_options(${target} PRIVATE ${extra} -Wall "$<$<NOT:$<COMPILE_LANGUAGE:CXX>>:-Wimplicit;-Wstrict-prototypes>")
    if(MINGW)
      target_compile_definitions(${target} PRIVATE USE_ICON MINGW=1)
      target_compile_options(${target} PRIVATE -fvisibility=hidden -ffunction-sections -flto -m64)
      if(NOT kind STREQUAL "native")
        target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-O2;-ffast-math>" "$<$<COMPILE_LANGUAGE:CXX>:-DCINTERFACE=>")
      else()
        target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-O2>")
      endif()
    else()
      target_compile_options(${target} PRIVATE -pipe "$<$<NOT:$<CONFIG:Debug>>:-O2;-fvisibility=hidden>")
    endif()
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Debug>:-DDEBUG;-D_DEBUG;-g;-O0>" "$<$<NOT:$<CONFIG:Debug>>:-DNDEBUG>")
    target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-std=c++20;-fno-exceptions;-fno-rtti>")
    if(kind STREQUAL "native")
      target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off -fno-strict-aliasing -fwrapv -fno-builtin -fPIC -U_GNU_SOURCE -D_DEFAULT_SOURCE ${NATIVE_WARNINGS})
    else()
      target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CXX_WARNINGS}>")
    endif()
    # These are compiler path maps, not post-build hash normalization.
    if(MINGW)
      target_compile_options(${target} PRIVATE "-fdebug-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/aftershock-source" "-fdebug-prefix-map=${CMAKE_BINARY_DIR}=/aftershock-source" "-fmacro-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}/=")
    else()
      target_compile_options(${target} PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}/=" "-fdebug-prefix-map=${CMAKE_BINARY_DIR}=${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
  endif()
endfunction()

foreach(module game cgame ui)
  string(TOUPPER "${module}" upper)
  if(NATIVE_${upper}_OBJECTS)
    set(${upper}_OBJECTS ${NATIVE_${upper}_OBJECTS})
    continue()
  endif()
  if(NOT BUILD_CLIENT AND NOT module STREQUAL "game")
    continue()
  endif()
  if(module STREQUAL "game")
    set(module_define QAGAME)
    set(prefix g)
  elseif(module STREQUAL "cgame")
    set(module_define CGAME)
    set(prefix cg)
  else()
    set(module_define UI)
    set(prefix ui)
  endif()
  foreach(source IN LISTS NATIVE_${upper}_SOURCES)
    get_filename_component(stem "${source}" NAME_WE)
    set(target native_${module}_${stem})
    add_library(${target} OBJECT game/module.cpp)
    engine_options(${target} native)
    string(SUBSTRING "${source}" 5 -1 relative)
    target_compile_definitions(${target} PRIVATE ${module_define} NATIVE_NAMESPACE=${module} NATIVE_SOURCE="${relative}")
    if(stem STREQUAL "${prefix}_main")
      target_compile_definitions(${target} PRIVATE NATIVE_EXPORTS="${module}/${prefix}_native_exports.inc")
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND stem STREQUAL "bg_lib")
      target_compile_definitions(${target} PRIVATE __NO_INLINE__)
    endif()
    list(APPEND ${upper}_OBJECTS $<TARGET_OBJECTS:${target}>)
  endforeach()
endforeach()

function(engine_link target)
  separate_arguments(extra_link NATIVE_COMMAND "${AFTERSHOCK_EXTRA_LINK_FLAGS}")
  target_link_options(${target} PRIVATE ${extra_link})
  target_link_libraries(${target} PRIVATE ${PLATFORM_LIBRARIES})
  if(MINGW)
    target_link_options(${target} PRIVATE -mwindows -Wl,--dynamicbase -Wl,--nxcompat -Wl,--gc-sections -fvisibility=hidden -flto)
  elseif(MSVC)
    target_link_options(${target} PRIVATE /MANIFEST:NO "$<$<CONFIG:Debug>:/DEBUG>")
  elseif(NOT APPLE)
    target_link_options(${target} PRIVATE -Wl,--gc-sections -fvisibility=hidden "$<$<CONFIG:Debug>:-rdynamic>")
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_link_options(${target} PRIVATE -Wl,--hash-style=both)
    endif()
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<LOWER_CASE:$<CONFIG>>-${AFTERSHOCK_PLATFORM}-${AFTERSHOCK_ARCH}")
endfunction()
